function mostrarTela(id) {
  document.querySelectorAll('.tela').forEach(t => t.classList.remove('ativa'));
  document.getElementById(id).classList.add('ativa');
  window.scrollTo(0,0);
}

function atualizarBreadcrumb(partes) {
  const bc = document.getElementById('breadcrumb');
  bc.innerHTML = partes.map((p, i) => {
    if (i < partes.length - 1) {
      return `<span onclick="${p.fn}">${p.label}</span> <span style="color:#444">›</span>`;
    }
    return `<span style="color:#f0f0f0;cursor:default">${p.label}</span>`;
  }).join(' ');
}

function irInicio() {
  sexoAtual = ''; nivelAtual = ''; fichaAtual = '';
  mostrarTela('tela-inicio');
  atualizarBreadcrumb([{label:'📍 Início', fn:'irInicio()'}]);
}

function irParaNivel(sexo) {
  sexoAtual = sexo;
  const label = sexo === 'M' ? '💪 Masculino' : '🏋️‍♀️ Feminino';
  document.getElementById('titulo-nivel').textContent = `Nível — ${label}`;
  mostrarTela('tela-nivel');
  atualizarBreadcrumb([
    {label:'📍 Início', fn:'irInicio()'},
    {label:label, fn:''}
  ]);
}

function irParaFichas(nivel) {
  nivelAtual = nivel;
  const chave = `${sexoAtual}_${nivel}`;
  const dados = treinos[chave];
  if (!dados) { alert('Treino não encontrado!'); return; }

  const nomes = {iniciante:'🌱 Iniciante', intermediario:'⚡ Intermediário', avancado:'🔥 Avançado'};
  const sexoLabel = sexoAtual === 'M' ? '💪 Masculino' : '🏋️‍♀️ Feminino';

  document.getElementById('titulo-fichas').textContent = `Fichas — ${nomes[nivel]}`;

  const cont = document.getElementById('container-fichas');
  cont.innerHTML = '';

  Object.entries(dados.fichas).forEach(([letra, ficha]) => {
    cont.innerHTML += `
      <div class="ficha-card" onclick="irParaExercicios('${letra}')">
        <div class="ficha-letra">${letra}</div>
        <div class="ficha-desc">${ficha.desc}</div>
        <div style="margin-top:8px;font-size:.75rem;color:var(--verde)">${ficha.exercicios.length} exercícios</div>
      </div>`;
  });

  mostrarTela('tela-fichas');
  atualizarBreadcrumb([
    {label:'📍 Início', fn:'irInicio()'},
    {label:sexoLabel, fn:`irParaNivel('${sexoAtual}')`},
    {label:nomes[nivel], fn:''}
  ]);
}

  const ficha = dados.fichas[letra];

  const badges = {iniciante:'iniciante', intermediario:'intermediario', avancado:'avancado'};
  const bTexts = {iniciante:'INICIANTE', intermediario:'INTERMEDIÁRIO', avancado:'AVANÇADO'};
  const nomes = {iniciante:'🌱 Iniciante', intermediario:'⚡ Intermediário', avancado:'🔥 Avançado'};
  const sexoLabel = sexoAtual === 'M' ? '💪 Masculino' : '🏋️‍♀️ Feminino';

  document.getElementById('titulo-ex').textContent = `Ficha ${letra} — ${ficha.desc}`;
  document.getElementById('badge-ex').innerHTML = `<span class="badge ${badges[nivelAtual]}">${bTexts[nivelAtual]}</span>`;

  const cont = document.getElementById('container-exercicios');
  cont.innerHTML = '';

  ficha.exercicios.forEach((ex, i) => {
    cont.innerHTML += `
    <div class="exercicio">
      <div class="ex-header" onclick="toggleEx(${i})">
        <div class="ex-num">${i+1}</div>
        <div class="ex-info">
          <div class="ex-nome">${ex.nome}</div>
          <div class="ex-musculo">${ex.musculo}</div>
        </div>
        <div class="ex-series">${ex.series}</div>
        <span class="ex-chevron" id="chev-${i}">▼</span>
      </div>
      <div class="ex-body" id="exb-${i}">
        <div class="ex-tags">
          <span class="tag verde">${ex.series}</span>
          <span class="tag">${ex.musculo}</span>
        </div>
        <div class="ex-video">
          <iframe src="${ex.video}" allowfullscreen loading="lazy"></iframe>
        </div>
        <h4 style="color:var(--verde);font-size:.85rem;margin:10px 0 6px">📋 Execução</h4>
        <ol class="ex-steps">
          ${ex.passos.map(p=>`<li>${p}</li>`).join('')}
        </ol>
        <div class="ex-dica"><strong>💡 Dica:</strong> ${ex.dica}</div>
        <div class="ex-erro"><strong>⚠️ Erro comum:</strong> ${ex.erro}</div>
      </div>
    </div>`;
  });

  mostrarTela('tela-exercicios');
  atualizarBreadcrumb([
    {label:'📍 Início', fn:'irInicio()'},
    {label:sexoLabel, fn:`irParaNivel('${sexoAtual}')`},
    {label:nomes[nivelAtual], fn:`irParaFichas('${nivelAtual}')`},
    {label:`Ficha ${letra}`, fn:''}
  ]);
}

function voltarNivel() { irParaNivel(sexoAtual); }
function voltarFichas() { irParaFichas(nivelAtual); }

function toggleEx(i) {
  const body = document.getElementById(`exb-${i}`);
  const chev = document.getElementById(`chev-${i}`);
  const aberto = body.classList.contains('aberto');
  body.classList.toggle('aberto', !aberto);
  chev.classList.toggle('aberto', !aberto);
}

function irMobilidade() {
  const cont = document.getElementById('container-mobilidade');
  cont.innerHTML = '';
  mobilidade.forEach(m => {
    cont.innerHTML += `
    <div class="mob-card">
      <h3>🧘 ${m.nome}</h3>
      <p>${m.desc}</p>
      <div class="mob-video">
        <iframe src="${m.video}" allowfullscreen loading="lazy"></iframe>
      </div>
    </div>`;
  });
  mostrarTela('tela-mobilidade');
  atualizarBreadcrumb([
    {label:'📍 Início', fn:'irInicio()'},
    {label:'🧘 Mobilidade', fn:''}
  ]);
}
